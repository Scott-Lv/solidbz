 //plane, Brillouin Zone
 function triangle(viewer) {                          <!--使用三角形函数--> 
        var vertices = [];                        <!--使用三角形函数绘制多面体-->
 //       var normals = [];

        vertices.push(new $3Dmol.Vector3( 0.000, 5.000, 0.000));     <!--定义布里渊区边界0点--> 
        vertices.push(new $3Dmol.Vector3( 4.330, 2.500, 0.000));     <!--定义布里渊区边界1点--> 
        vertices.push(new $3Dmol.Vector3( 4.330,-2.500, 0.000));     <!--定义布里渊区边界2点--> 
        vertices.push(new $3Dmol.Vector3( 0.000,-5.000, 0.000));     <!--定义布里渊区边界3点--> 
        vertices.push(new $3Dmol.Vector3(-4.330,-2.500, 0.000));     <!--定义布里渊区边界4点--> 
        vertices.push(new $3Dmol.Vector3(-4.330, 2.500, 0.000));     <!--定义布里渊区边界5点--> 

        var faces1 = [ 0,1,2];                       <!--定义使用0，1和2绘制三角形平面-->    
        var faces2 = [ 0,2,3];
        var faces3 = [ 0,3,4]; 
        var faces4 = [ 0,4,5]; 

        var spec1 = {vertexArr:vertices, faceArr:faces1, color:'red', opacity: 0.5};     <!--定义三角形平面faces1的颜色和透明度-->  
 	   var spec2 = {vertexArr:vertices, faceArr:faces2, color:'red', opacity: 0.5};
        var spec3 = {vertexArr:vertices, faceArr:faces3, color:'red', opacity: 0.5};
 	   var spec4 = {vertexArr:vertices, faceArr:faces4, color:'red', opacity: 0.5};

        viewer.addCustom(spec1);                                 <!--按照spec1的定义显示平面-->  
 	   viewer.addCustom(spec2);
        viewer.addCustom(spec3);
 	   viewer.addCustom(spec4);	
     }
    triangle(viewer);
//      viewer.zoomTo();
  viewer.render();                                            <!--显示生效-->  

<!--绘制K点及其路径-->
